## 🎯 COMPLETE BOOK WRITING EXAMPLE

### Novel Writing Pipeline

```python
from forge_book import BookForge

# Initialize THE FORGE Book Writing System
forge = BookForge()

# STEP 1: Create book concept
concept = forge.create_concept(
    genre="Mystery Thriller",
    target_audience="Adult",
    themes=["Justice", "Redemption", "Truth"],
    setting="Modern-day San Francisco",
    word_count_target=85000
)

# STEP 2: Develop characters
protagonist = forge.create_character(
    name="Detective Sarah Chen",
    role="Protagonist",
    age=38,
    background="Former FBI profiler",
    flaw="Trusts no one after betrayal",
    goal="Solve serial killer case",
    arc="Learn to trust again"
)

antagonist = forge.create_character(
    name="The Collector",
    role="Antagonist",
    motivation="Punish the corrupt",
    secret="Actually victim of corruption",
    twist="Sympathetic despite crimes"
)

# STEP 3: Generate outline
outline = forge.generate_outline(
    concept=concept,
    characters=[protagonist, antagonist],
    structure="Three-act with twist",
    chapters=30
)

# STEP 4: Write the book (HIGHEST QUALITY)
book = forge.write_book(
    outline=outline,
    quality_level="PUBLISHING_PROFESSIONAL",
    style="Fast-paced, atmospheric",
    pov="Third-person limited",
    tense="Past tense"
)

# STEP 5: Quality upscale
enhanced = forge.upscale_quality(
    book=book,
    target_quality="AWARD_WINNING"
)

# STEP 6: Edit and polish
polished = forge.professional_edit(
    manuscript=enhanced,
    editing_passes=[
        "developmental",
        "content",
        "copy",
        "line",
        "proofread"
    ]
)

# STEP 7: Format for publishing
final = forge.format_for_publishing(
    manuscript=polished,
    formats=["PDF", "EPUB", "MOBI"],
    include_front_matter=True,
    include_back_matter=True
)

# STEP 8: Generate marketing materials
marketing = forge.create_marketing(
    book=final,
    materials=[
        "back_cover_copy",
        "author_bio",
        "press_release",
        "social_media_posts",
        "email_announcement"
    ]
)

# STEP 9: Sequel planning
sequel = forge.plan_sequel(
    book=final,
    unresolved_threads=True,
    character_arcs_continued=True
)

print("✅ Book complete and ready to publish!")
print(f"Title: {final.title}")
print(f"Word count: {final.word_count}")
print(f"Quality score: {final.quality_score}/100")
print(f"Ready for: {final.publishing_channels}")
print(f"Sequel potential: {sequel.potential}")
```
