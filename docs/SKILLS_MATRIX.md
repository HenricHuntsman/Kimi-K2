        # Database
        ".sql", ".db", ".sqlite", ".mdb",
        
        # Config
        ".conf", ".config", ".cfg", ".properties",
        
        # ROM files (for gaming)
        ".gb", ".gbc", ".gba", ".nds", ".3ds", ".nes",
        ".snes", ".n64", ".z64",
        
        # And many more...
        ".*"  # Literally ANY file
    ]
    
    def upload_file(self, file_path, destination):
        """
        Upload ANY file type
        """
        file_type = self.detect_file_type(file_path)
        
        if file_type in ["code", "text"]:
            return self.upload_text_file(file_path, destination)
        elif file_type == "binary":
            return self.upload_binary_file(file_path, destination)
        elif file_type == "image":
            return self.upload_image_file(file_path, destination)
        elif file_type == "archive":
            return self.upload_archive(file_path, destination)
        else:
            return self.upload_raw(file_path, destination)
    
    def analyze_file(self, file_path):
        """
        Analyze any file and extract information
        """
        analyses = {
            "code": self.analyze_code,
            "document": self.analyze_document,
            "image": self.analyze_image,
            "data": self.analyze_data,
            "binary": self.analyze_binary
        }
        
        file_type = self.detect_file_type(file_path)
        analyzer = analyses.get(file_type, self.analyze_raw)
        
        return analyzer(file_path)
